// 🔴 修正：掃描時明確帶入 &timeframe=Day 參數，避免計算邏輯歧義 async function startScanner() { const target = document.getElementById('scanTarget').value; const progress = document.getElementById('scanProgress'); const container = document.getElementById('scanResultContainer'); const tableBody = document.getElementById('scanTableBody'); const rr = document.getElementById('rrRatio').value; progress.style.display = 'block'; container.style.display = 'none'; tableBody.innerHTML = ''; let matchedStocks = []; for (let i = 0; i < WATCH_POOL.length; i++) { const code = WATCH_POOL[i]; progress.innerText = `🔍 正在精準分析台股指標池 (${i + 1}/${WATCH_POOL.length})：${code}...`; try { // 帶入 &timeframe=Day 確保與個股單獨查詢邏輯完全一致 const res = await fetch(`${WORKER_URL}/?stock_id=${code}&rr=${rr}&timeframe=Day`); const data = await res.json(); if (data && !data.error) { let isMatch = false; if (target === 'STRONG_BUY' && data.score >= 75) isMatch = true; else if (target === 'BUY' && data.score >= 60) isMatch = true; else if (target === 'SELL' && data.score <= 40) isMatch = true; if (isMatch) matchedStocks.push(data); } } catch (e) { console.error(`掃描 ${code} 失敗:`, e); } await new Promise(resolve => setTimeout(resolve, 300)); } progress.style.display = 'none'; container.style.display = 'block'; if (matchedStocks.length === 0) { tableBody.innerHTML = `目前市場無符合「${document.getElementById('scanTarget').selectedOptions[0].text}」條件之標的。`; return; } matchedStocks.forEach(st => { const tr = document.createElement('tr'); tr.className = 'scan-row'; const nameText = st.stock_name ? `${st.stock_id} ${st.stock_name}` : st.stock_id; tr.innerHTML = ` ${nameText} ${st.score} 分 $${st.trade_plan.entry} ${st.signal} `; tableBody.appendChild(tr); }); }
